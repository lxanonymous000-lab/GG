local RAYCAST_SCHEMA = {
    ArgCountRequired = 3,
    Args = { "Instance", "Vector3", "Vector3", "RaycastParams" },

local function getChar(player)
    local folder = Workspace:FindFirstChild(DAHOOD_CHAR_FOLDER)
    if folder then
        local m = folder:FindFirstChild(player.Name)
        if m then return m end
    end
    return player.Character
end

local function getVelocity(hrp)
    return hrp.AssemblyLinearVelocity ~= nil and hrp.AssemblyLinearVelocity or hrp.Velocity
end

local function isAlive(root)
    if not root or not root.Parent then return false end
    if not root:IsDescendantOf(Workspace) then return false end
    local h = root.Parent:FindFirstChild("Humanoid")
    return h and h.Health > 0
end

local function isKnockedOut(root)
    if not root or not root.Parent then return false end
    local h = root.Parent:FindFirstChild("Humanoid")
    if not h then return false end
    return h.Health <= 0
end

local function isFriend(player)
    local ok, result = pcall(function()
        return LocalPlayer:IsFriendsWith(player.UserId)
    end)
    return ok and result
end

local function hasLoS(root)
    if not root then return false end
    local origin = Camera.CFrame.Position
    local dir    = root.Position - origin
    local rp     = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    local chars = {}
    for _, p in Players:GetPlayers() do
        local c = getChar(p)
        if c then table.insert(chars, c) end
    end
    if LocalPlayer.Character then table.insert(chars, LocalPlayer.Character) end
    rp.FilterDescendantsInstances = chars
    return Workspace:Raycast(origin, dir, rp) == nil
end

local function buildIncludeParams()
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Include
    local chars = {}
    for _, p in Players:GetPlayers() do
        if p ~= LocalPlayer then
            local c = getChar(p)
            if c then table.insert(chars, c) end
        end
    end
    rp.FilterDescendantsInstances = chars
    return rp
end

local function validateArgs(args, schema)
    if #args < schema.ArgCountRequired then return false end
    local n = 0
    for i, v in next, args do
        if typeof(v) == schema.Args[i] then n = n + 1 end
    end
    return n >= schema.ArgCountRequired
end

local function calcDirection(origin, target, spread)
    local dir
    if spread and spread > 0 then
        local v186 = spread * (target - origin).Magnitude / 100
        local lim  = math.floor(v186 * 1000)
        if lim < 1 then lim = 1 end
        local jitter = Vector3.new(
            math.random(-lim, lim) / 1000,
            math.random(-lim, lim) / 1000,
            math.random(-lim, lim) / 1000
        )
        dir = ((target + jitter) - origin).Unit
    else
        dir = (target - origin).Unit
    end
    return dir * 1000
end

local function passesChecks(player, root)
    if cfg.friendCheck  and isFriend(player)    then return false end
    if cfg.knockedCheck and isKnockedOut(root)  then return false end
    if cfg.wallCheck    and not hasLoS(root)    then return false end
    return true
end

local function findClosest()
    local best, bestDist = nil, math.huge
    local cx = Camera.ViewportSize.X / 2
    local cy = Camera.ViewportSize.Y / 2
    for _, p in Players:GetPlayers() do
        if p ~= LocalPlayer then
            local char = getChar(p)
            if char then
                local root = char:FindFirstChild("HumanoidRootPart")
                local hum  = char:FindFirstChild("Humanoid")
                if root and hum and hum.Health > 0 and passesChecks(p, root) then
                    local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
                    if onScreen then
                        local d = math.sqrt((pos.X-cx)^2 + (pos.Y-cy)^2)
                        if (cfg.fovRadius == 0 or d <= cfg.fovRadius) and d < bestDist then
                            bestDist = d
                            best     = root
                        end
                    end
                end
            end
        end
    end
    return best
end

local function camMoved()
    local dot = Camera.CFrame.LookVector:Dot(lastCamCFrame.LookVector)
    return math.acos(math.clamp(dot, -1, 1)) > camThreshold
end

RunService.RenderStepped:Connect(function()
    aimTarget = findClosest()

    if not cfg.camlockEnabled then return end

    if camMoved() then
        lockedTarget  = findClosest()
        lastCamCFrame = Camera.CFrame
        return
    end

    if not isAlive(lockedTarget) then
        lockedTarget = findClosest()
        lastCamCFrame = Camera.CFrame
        if not lockedTarget then return end
    end

    local root = lockedTarget
    if not root or not root.Parent then return end

    local head   = root.Parent:FindFirstChild("Head")
    local vel    = getVelocity(root)
    local pred   = Vector3.new(vel.X * cfg.predictionH, vel.Y * cfg.predictionV, vel.Z * cfg.predictionH)
    local aimPos = (head and head.Position or root.Position) + pred
    Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, aimPos), cfg.smoothness)
    lastCamCFrame = Camera.CFrame
end)

local function installHook()
    if hookRef then return end
    hookRef = hookmetamethod(game, "__namecall", newcclosure(function(...)
        local self   = ...
        local method = getnamecallmethod()
        local args   = {...}
        if  self == Workspace
        and method == "Raycast"
        and not checkcaller()
        and cfg.silentAimEnabled
        and aimTarget
        and validateArgs(args, RAYCAST_SCHEMA)
        then
            local origin = args[2]
            local spread = (equippedTool == "Double Barrel" or equippedTool == "Tactical Shotgun") and 6 or nil
            args[3] = calcDirection(origin, aimTarget.Position, spread)
            args[4] = buildIncludeParams()
            return hookRef(unpack(args))
        end
        return hookRef(...)
    end))
end

local function onCharAdded(char)
    equippedTool = nil
    char.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then
            equippedTool = child.Name
            if not FIREARM_BLACKLIST[equippedTool] then
                lockedTarget  = findClosest()
                lastCamCFrame = Camera.CFrame
            end
        end
    end)
    char.ChildRemoved:Connect(function(child)
        if child:IsA("Tool") then equippedTool = nil end
    end)
end

if LocalPlayer.Character then onCharAdded(LocalPlayer.Character) end
LocalPlayer.CharacterAdded:Connect(onCharAdded)
installHook()

    menuFrame.Visible = true
    if menuOpen then
        tw(menuFrame, {Size = UDim2.new(0, 260, 0, MENU_H)}, 0.28, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
        tw(settingsBtn, {TextColor3 = Color3.fromRGB(255,255,255)}, 0.16)
    else
        tw(menuFrame, {Size = UDim2.new(0, 260, 0, 0)}, 0.22, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
        tw(settingsBtn, {TextColor3 = Color3.fromRGB(180,180,180)}, 0.16)
        task.delay(0.24, function()
            if not menuOpen then menuFrame.Visible = false end
        end)
    end
end

local function setMainVisual(on)
    tw(toggleBtn, {TextColor3 = on and Color3.fromRGB(255,255,255) or Color3.fromRGB(200,200,200)}, 0.18)
    toggleBtn.Text = on and "  Aruxia  ·  ON" or "  Aruxia  ·  OFF"
    tw(glowBar, {BackgroundTransparency = on and 0.18 or 0.52}, 0.2)
end

toggleBtn.MouseButton1Click:Connect(function()
    local on = not (cfg.camlockEnabled or cfg.silentAimEnabled)
    cfg.camlockEnabled   = on
    cfg.silentAimEnabled = on
    if not on then lockedTarget = nil end
    setMainVisual(on)
end)
